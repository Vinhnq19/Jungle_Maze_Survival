using System.Diagnostics;
var seconds = args.Length > 0 ? int.Parse(args[0]) : 900;
var count = args.Length > 1 ? int.Parse(args[1]) : 5;
var leaveAfter = args.Length > 2 ? int.Parse(args[2]) : -1; // seconds after which bot #0 disconnects (to force a real elimination)
var bots = new List<MatchNetworkClient>();
for (var i = 0; i < count; i++)
{
    var b = new MatchNetworkClient("Udp") { AutoReady = true };
    await b.ConnectAsync("127.0.0.1", 7777, "ws://127.0.0.1:7781/game", "ROOM42");
    bots.Add(b);
}
var clock = Stopwatch.StartNew(); double nextInput = 0; double nextLog = 0; var left = false;
Console.WriteLine($"[bot] {count} bots connecting");
while (clock.Elapsed.TotalSeconds < seconds)
{
    foreach (var b in bots) b.Poll();
    var now = clock.Elapsed.TotalSeconds;
    if (now >= nextInput) { foreach (var b in bots) b.Input(0.0f, 0.0f); nextInput = now + 1.0 / 30; }
    if (leaveAfter >= 0 && !left && now >= leaveAfter) { left = true; Console.WriteLine("[bot] bot0 disconnects now"); await bots[0].DisposeAsync(); bots.RemoveAt(0); }
    if (now >= nextLog) { var b = bots[0]; Console.WriteLine($"[bot] t={now:F0}s bots={bots.Count} phase={b.Phase} snaps={b.SnapshotCount} rejected={b.Rejected} netId={b.OwnNetId} hasResult={b.Result != null}"); nextLog = now + 5; }
    Thread.Sleep(1);
}
